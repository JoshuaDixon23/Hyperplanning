@startuml

skinparam linetype ortho
skinparam nodesep 90
skinparam ranksep 90
skinparam padding 12
skinparam classAttributeIconSize 0

' ─────────────────────────────────────────────
'  USERS
' ─────────────────────────────────────────────
abstract class User {
    firstName
    lastName
    emailUniv
}

abstract class Student {
    emailPersonal
}

class Professor
class StaffDRI
class Admin
class LocalStudent
class InternationalStudent

' ─────────────────────────────────────────────
'  ACADEMIC STRUCTURE
' ─────────────────────────────────────────────
class Promo {
    Enum studyLevel
}

class Module {
    code
    name
    language
    ECTS
}

class UFR {
    name
}

class Group {
    Enum type - promo, TD, TP
}

class Course {
    date
    startTime
    duration
    courseType
    Map courseHours
}

' ─────────────────────────────────────────────
'  BUILDINGS
' ─────────────────────────────────────────────
class Building {
    name
    localisation
    Map openingHours
}

class Room {
    num
    capacity
    Enum type - TP, amphitheater etc
}

class Campus {
    city
    image
}

' ─────────────────────────────────────────────
'  BASKET / ENROLLMENT
' ─────────────────────────────────────────────
class BasketModule
class BasketFinal
class BasketEntry <<association>>

' ─────────────────────────────────────────────
'  LAYOUT — paysage réel
' ─────────────────────────────────────────────

' Ligne 1 : utilisateurs bien étalés
LocalStudent         -[hidden]r- Student
Student              -[hidden]r- InternationalStudent
InternationalStudent -[hidden]r- User
User                 -[hidden]r- Professor
Professor            -[hidden]r- Admin
Admin                -[hidden]r- StaffDRI

' Ligne 2 : coeur académique horizontal
Promo                -[hidden]r- Group
Module               -[hidden]r- Course
Course               -[hidden]r- UFR
UFR                  -[hidden]r- Building
Building             -[hidden]r- Room
Room                 -[hidden]r- Campus

' Ligne 3 : paniers placés sous la zone centrale/droite
BasketModule         -[hidden]r- BasketFinal

' Alignements verticaux
LocalStudent         -[hidden]d- Promo
Student              -[hidden]d- Group
User                 -[hidden]d- Module
Professor            -[hidden]d- Course
Admin                -[hidden]d- UFR
StaffDRI             -[hidden]d- Building

Module               -[hidden]d- BasketModule
UFR                  -[hidden]d- BasketFinal

' ─────────────────────────────────────────────
'  INHERITANCE
' ─────────────────────────────────────────────
Professor            -u-|> User
StaffDRI             -u-|> User
Admin                -u-|> User
Student              -u-|> User
LocalStudent         -u-|> Student
InternationalStudent -u-|> Student

' ─────────────────────────────────────────────
'  ASSOCIATIONS
' ─────────────────────────────────────────────

' Professor
Professor "responsible"      -d- "0..1" Promo
Professor "responsible"      -d-> "0..*" Module
Professor "1..*"             -d- "*"    Course : teaches >

' LocalStudent
LocalStudent       <-d-o Promo
LocalStudent "*"   <-d-  "*" Group : form >

' Module / Course / Group
Module             -d->  "0..*" Course : describes >
Group "1..*"       <-r-  "1..*" Module : follows >
Group "*"          -r-  "*"    Course : follows >
Promo              -d->  "1..*" Group  : divide <

' UFR / Building
UFR                *-l-> Promo
Building           *-d-> Room
Building           <-r-* Campus
Building "1..*"    <-l-  "0..1" UFR : belongs >
UFR             "*"<- Campus : locate <

' Room / Course
Room "0..1"        -u-> "*" Course

' Basket
BasketModule       o-u- Module
BasketModule       <-l- InternationalStudent
BasketFinal        *--  BasketEntry 
BasketFinal        -u-* InternationalStudent

' ModuleGroup association class
'(Group, Module) .. ModuleGroup
BasketEntry        <-u- Group
BasketEntry        <-u- Module

' UFR / Admin
UFR                <-u- Admin

@enduml
