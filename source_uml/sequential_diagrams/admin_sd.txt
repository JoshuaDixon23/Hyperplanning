@startuml
header Nominal Scenario - Administrator

actor Admin
participant "App" as App
database "Database" as DB

group Authentication
    Admin -> App: Authentication Request (Logins)
    App -> DB: Verify credentials
    DB --> App: Admin profile validated
    App --> Admin: Access to Dashboard
end

group UFR Management
    Admin -> App: Create "Sciences and Technologies" UFR
    App -> DB: Save new UFR
    Admin -> App: Generate "Computer Science - L3"  Program within UFR
    App -> DB: Associate Program with UFR
    Admin -> App: Configure parameters (Academic year, Expected headcount)
    App -> DB: Update Database
end

group Module Management
    Admin -> App: Add modules ("Crypto", "Databases")
    App -> DB: Link modules to L3 Program
    App --> Admin: Modules registered

group Course Management
    Admin -> App: Create new course acnd add to a module
    App -> DB: Link course and Crypto module
    App --> Admin: Display new course
end
end

group Teacher Management
    Admin -> App: Change teacher name Grey to Sheperd
    App -> DB: Apply changes
    App --> Admin: Display new name
end

group Student Management
    Admin -> App: Add new student Josh
    App -> DB: Insert Josh
    App --> Admin: Display new student
end


group Map Management
    Admin -> App: Modify map (Add new building)
    App -> DB: Update map
    App --> Admin: Map updated
end

group Menu Updates
    Admin -> App: Update Canteen Menu
    App -> DB: Save weekly menu
    App --> Admin: Menu published successfully
end

@enduml
