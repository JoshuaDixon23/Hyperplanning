  [*] --> Editing
  Editing --> Editing : (startTime, duration, room, group)
}

  [*] --> PendingReview
  PendingReview --> PendingReview : waiting for the admin
}

ModificationRequested --> Published : [when time before cours < 24h] 



Modified --> Published : update CourseReal



Published --> InProgress : [time == startTime]
InProgress --> Completed : [time >= startTime + duration]


Completed --> [*]
