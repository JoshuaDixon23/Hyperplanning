title Professor - Nominal Scenario (Course info via planning)

actor "admin" as A
boundary "App" as UI
database "Database" as DB
group Authentication
    P -> UI : Provide login credentials
    UI -> DB : Verify user ID and password
    DB --> UI : Authentication successful
    UI --> P : Access granted
end

group  Consult planning
    P -> UI : Request schedule
    UI -> DB : Retrieve schedule data
    DB --> UI : Schedule data
    UI --> P : Display schedule
end
